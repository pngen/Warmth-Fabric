add_executable(warmth_cli warmth_cli.cpp)
target_link_libraries(warmth_cli PRIVATE warmth)
if(WIN32)
    target_link_libraries(warmth_cli PRIVATE ws2_32)
endif()
target_compile_features(warmth_cli PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(warmth_cli PRIVATE /W4 /WX /permissive- /Zc:__cplusplus /EHsc)
else()
    target_compile_options(warmth_cli PRIVATE -Wall -Wextra -Werror)
endif()
if(WIN32)
    target_compile_definitions(warmth_cli PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
